💻 Usage Example

Here's how you would use this module in your own project:

// Import the enhanced API import { createEnhancedFarertAPI } from './enhanced-sdk-module.js'; // Initialize the API const api = await createEnhancedFarertAPI(); // Search for a station with SDK validation const stationResult = await api.searchStation('東京'); console.log(stationResult); // Build a route with enhanced formatting const routeResult = await api.buildRoute({ start: '東京', via: '東海道線', end: '横浜' }); console.log(routeResult); // Get performance metrics const metrics = api.getPerformanceMetrics(); console.log(`API calls: ${metrics.apiCalls}`);
